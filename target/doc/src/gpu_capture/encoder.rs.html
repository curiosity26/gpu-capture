3pub struct VideoEncoder {
4    encoder: ffmpeg::encoder::Video,
5    output: ffmpeg::format::context::Output,

8impl VideoEncoder {
9    pub fn new(path: &str, w: u32, h: u32, fps: u32, bitrate: i32) -> Result<Self, String> {
10        ffmpeg::init().map_err(|e| format!("FFmpeg init failed: {:?}", e))?;
11
12        let codec = ffmpeg::encoder::find(ffmpeg::codec::Id::H264)
13            .ok_or("H.264 encoder not found")?;
14
15        let mut enc = ffmpeg::codec::context::Context::new_with_codec(codec)
16            .encoder()
17            .video()
18            .map_err(|e| format!("Failed to create video encoder: {:?}", e))?;
19
20        enc.set_width(w);
21        enc.set_height(h);
22        enc.set_time_base(ffmpeg::Rational::new(1, fps as i32));
23        enc.set_frame_rate(Some(ffmpeg::Rational::new(fps as i32, 1)));
24        enc.set_format(ffmpeg::format::Pixel::YUV420P);
25
26        if bitrate > 0 {
27            enc.set_bit_rate(bitrate as usize);
28        }
29
30        let opened = enc.open()
31            .map_err(|e| format!("Failed to open encoder: {:?}", e))?;
32
33        let mut output = ffmpeg::format::output(path)
34            .map_err(|e| format!("Failed to create output file: {:?}", e))?;
35
36        let mut stream = output.add_stream(opened.codec().ok_or("No codec")?)
37            .map_err(|e| format!("Failed to add stream: {:?}", e))?;
38        stream.set_time_base(ffmpeg::Rational::new(1, fps as i32));
39
40        output.write_header()
41            .map_err(|e| format!("Failed to write header: {:?}", e))?;
42
43        Ok(Self { encoder: opened, output })
44    }
45
46    pub fn encode_frame(&mut self, frame: &ffmpeg::frame::Video) -> Result<(), String> {
47        self.encoder.send_frame(frame)
48            .map_err(|e| format!("Failed to send frame: {:?}", e))?;
49
50        let mut packet = ffmpeg::Packet::empty();
51        while self.encoder.receive_packet(&mut packet).is_ok() {
52            packet.set_stream(0);
53            packet.write_interleaved(&mut self.output)
54                .map_err(|e| format!("Failed to write packet: {:?}", e))?;
55        }
56
57        Ok(())
58    }
59
60    pub fn flush(mut self) -> Result<(), String> {
61        self.encoder.send_eof()
62            .map_err(|e| format!("Failed to flush encoder: {:?}", e))?;
63
64        let mut packet = ffmpeg::Packet::empty();
65        while self.encoder.receive_packet(&mut packet).is_ok() {
66            packet.set_stream(0);
67            packet.write_interleaved(&mut self.output)
68                .map_err(|e| format!("Failed to write packet: {:?}", e))?;
69        }
70
71        self.output.write_trailer()
72            .map_err(|e| format!("Failed to write trailer: {:?}", e))?;
73
74        Ok(())
75    }
